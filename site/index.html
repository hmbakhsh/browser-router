config.jsonthe whole source of truth
{
  "defaultProfile": "Personal",
  "rules": [{
    "name": "Acme stuff",
    "profile": "Work",
    "enabled": true,
    "match": {
      "host": "github.com",
      "pathPrefix": "/acme"
    }
  }]
}
